cmake_minimum_required(VERSION 3.16)
project(zfc_bringup LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
install(DIRECTORY config launch urdf DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(configuration_test test/test_configuration.py)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  find_package(hardware_interface REQUIRED)
  find_package(controller_interface REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rclcpp REQUIRED)
  ament_add_gtest(si_calibration_test test/si_calibration_test.cpp)
  target_include_directories(si_calibration_test PRIVATE ../zfc_ethercat_hardware/include)
  ament_add_gtest(control_resources_test test/control_resources_test.cpp)
  ament_target_dependencies(control_resources_test hardware_interface rclcpp controller_interface pluginlib)
  target_compile_definitions(control_resources_test PRIVATE ZFC_BRINGUP_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
  option(ZFC_HARDWARE_TESTS "Build fake-IgH protection regressions (requires real IgH SDK)" OFF)
  if(ZFC_HARDWARE_TESTS)
    find_package(zfc_ethercat_core REQUIRED)
    ament_add_gtest(plugins_test test/plugins_test.cpp test/fake_igh.cpp)
    ament_target_dependencies(plugins_test hardware_interface controller_interface pluginlib ament_index_cpp)
    target_link_libraries(plugins_test zfc_ethercat_core::zfc_ethercat_core)
    target_compile_definitions(plugins_test PRIVATE ZFC_BRINGUP_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
    # Test-only interposition. Production still requires genuine IgH.
    target_link_options(plugins_test PRIVATE -Wl,--export-dynamic)
  endif()
endif()
ament_package()
