cmake_minimum_required(VERSION 3.16)
project(zfc_calibration_analysis)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/calibration_analysis scripts/analyze_trial DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(analysis_test test/test_analysis.py TIMEOUT 30)
  ament_add_pytest_test(coordinator_test test/test_coordinator.py TIMEOUT 45)
endif()
ament_package()
