cmake_minimum_required(VERSION 3.16)
project(zfc_ethercat_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(cmake/igh.cmake)
add_library(${PROJECT_NAME} SHARED src/cia402.cpp src/clearpath_pdo.cpp src/elm3604_pdo.cpp src/ethercat_system.cpp)
add_library(zfc_ethercat_core::zfc_ethercat_core ALIAS ${PROJECT_NAME})
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/zfc_ethercat_core>)
target_link_libraries(${PROJECT_NAME} PUBLIC EtherLab::ethercat)
# Retain the discovered non-system IgH library directory in the installed core.
set_target_properties(${PROJECT_NAME} PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
if(NOT ZFC_ROOT_BUILD)
  find_package(ament_cmake REQUIRED)
  install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
  install(DIRECTORY include/ DESTINATION include/zfc_ethercat_core)
  ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
endif()
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_link_libraries(core_test PRIVATE ${PROJECT_NAME})
  add_test(NAME core_test COMMAND core_test)
endif()
if(NOT ZFC_ROOT_BUILD)
  ament_package(CONFIG_EXTRAS cmake/igh.cmake)
endif()
