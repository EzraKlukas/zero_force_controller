cmake_minimum_required(VERSION 3.16)
project(zfc_profiling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(controller_manager 2.54 REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(rclcpp REQUIRED)
find_package(zfc_timing REQUIRED)
add_executable(profile_control_node src/profile_control_node.cpp)
ament_target_dependencies(profile_control_node controller_manager realtime_tools rclcpp)
target_link_libraries(profile_control_node zfc_timing::zfc_timing)
install(TARGETS profile_control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
