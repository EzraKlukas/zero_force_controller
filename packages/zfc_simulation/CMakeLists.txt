cmake_minimum_required(VERSION 3.16)
project(zfc_simulation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
add_library(zfc_world_force SHARED src/world_force.cpp)
target_link_libraries(zfc_world_force PRIVATE ignition-gazebo6::ignition-gazebo6)
target_include_directories(zfc_world_force PRIVATE include)
target_compile_options(zfc_world_force PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS zfc_world_force LIBRARY DESTINATION lib)
install(PROGRAMS scripts/simulation_ready.py DESTINATION lib/${PROJECT_NAME} RENAME simulation_ready)
install(PROGRAMS scripts/force_input.py DESTINATION lib/${PROJECT_NAME} RENAME force_input)
install(PROGRAMS scripts/plot_relay.py DESTINATION lib/${PROJECT_NAME} RENAME plot_relay)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(force_command_test test/force_command_test.cpp)
  target_include_directories(force_command_test PRIVATE include)
endif()
ament_package()
