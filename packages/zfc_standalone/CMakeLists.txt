cmake_minimum_required(VERSION 3.16)
project(zfc_standalone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
if(NOT ZFC_ROOT_BUILD)
  find_package(ament_cmake REQUIRED)
  find_package(zfc_ethercat_core REQUIRED)
endif()
add_executable(zero_force_controller src/main.cpp src/drive_logic.cpp)
target_include_directories(zero_force_controller PRIVATE include)
target_link_libraries(zero_force_controller PRIVATE zfc_ethercat_core::zfc_ethercat_core Boost::headers Threads::Threads)
target_compile_options(zero_force_controller PRIVATE -Wall -Wextra -Wpedantic)
add_executable(profiler_bench src/profiler_bench.cpp)
target_link_libraries(profiler_bench PRIVATE zfc_ethercat_core::zfc_ethercat_core)
add_executable(matched_hold_runner src/matched_main.cpp)
target_link_libraries(matched_hold_runner PRIVATE zfc_ethercat_core::zfc_ethercat_core Threads::Threads)
target_compile_options(matched_hold_runner PRIVATE -Wall -Wextra -Wpedantic)
if(ZFC_ROOT_BUILD)
  set_target_properties(matched_hold_runner profiler_bench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  set_target_properties(zero_force_controller PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
else()
  install(TARGETS zero_force_controller matched_hold_runner profiler_bench DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
