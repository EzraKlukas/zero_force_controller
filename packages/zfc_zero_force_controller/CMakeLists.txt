cmake_minimum_required(VERSION 3.16)
project(zfc_zero_force_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(zfc_timing REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(${PROJECT_NAME} SHARED src/zero_force_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} controller_interface pluginlib rclcpp_lifecycle)
target_link_libraries(${PROJECT_NAME} zfc_timing::zfc_timing)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(calibration_sequencer_test test/calibration_sequencer_test.cpp)
  target_link_libraries(calibration_sequencer_test ${PROJECT_NAME})
  ament_add_gtest(controller_test test/controller_test.cpp)
  target_link_libraries(controller_test ${PROJECT_NAME})
  ament_target_dependencies(controller_test controller_interface rclcpp_lifecycle)
endif()
ament_package()
